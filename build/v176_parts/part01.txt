/* War Room v1.7.26 - Atomic per-character Inspect snapshots */
(function(){
  const norm=s=>String(s||'').trim(), key=s=>norm(s).toLowerCase();
  let timer=null,activeScope=null,activeName='',requestToken=0;
  const snapshots=new Map(),inflight=new Map();
  function rosterRef(){try{if(typeof roster!=='undefined'&&Array.isArray(roster))return roster}catch(e){}return Array.isArray(window.roster)?window.roster:[]}
  function rosterByName(n){return rosterRef().find(p=>key(p?.name)===key(n))||null}
  function gearRows(p){return Array.isArray(p?.gear)?p.gear:Array.isArray(p?.equipment)?p.equipment:Array.isArray(p?.items)?p.items:[]}
  function findExact(root,text){if(!root?.isConnected)return null;return [...root.querySelectorAll('*')].find(el=>norm(el.textContent).toUpperCase()===text.toUpperCase())||null}
  function dossier(){for(const sel of ['[role="dialog"]','.wr-dossier-fixed','.dossier','.modal','.drawer'])for(const el of document.querySelectorAll(sel))if(el.isConnected&&findExact(el,'AVG ILVL')&&findExact(el,'ROLE / SPEC'))return el;return null}
  function detectName(scope){if(!scope?.isConnected)return'';const names=new Set(rosterRef().map(p=>norm(p?.name)).filter(Boolean));for(const sel of ['h1','h2','h3','h4'])for(const el of scope.querySelectorAll(sel)){const t=norm(el.textContent);if(names.has(t))return t}const leaves=[...scope.querySelectorAll('strong,b,div,span')].filter(el=>!el.children.length);for(const el of leaves){const t=norm(el.textContent);if(names.has(t))return t}return''}
  function talentText(p){const t=p?.talentsData??p?.talents;if(!t)return'Armory data unavailable';if(typeof t==='string'){try{return talentText({talentsData:JSON.parse(t)})}catch(e){return t}}if(Array.isArray(t))return t.map(x=>`${norm(x?.name||x?.tree||x?.spec||'?')} ${x?.points??x?.value??''}`.trim()).filter(Boolean).join(' / ');if(typeof t==='object')return Object.entries(t).map(([k,v])=>typeof v==='object'&&v?`${norm(v.name||v.tree||k)} ${v.points??v.value??''}`.trim():`${k} ${v}`).join(' / ');return'Talents loaded'}
  function snapshotFromRecord(record,name){const base=rosterByName(name)||{},items=Array.isArray(record?.items)?record.items:[];return {...base,name:norm(record?.name||name),spec:norm(record?.activeSpec||record?.spec||base.spec),talentSpec:norm(record?.activeSpec||record?.spec||base.talentSpec),activeSpec:norm(record?.activeSpec||record?.spec||base.activeSpec),secondarySpec:norm(record?.secondarySpec||base.secondarySpec),role:base.role||'',ilvl:record?.ilvl??base.ilvl??null,gearscore:record?.gearscore??base.gearscore??null,talentsData:record?.talents??base.talentsData??base.talents??null,gear:items,equipment:items,items,race:norm(record?.race||base.race),gender:record?.gender??base.gender??null,appearance:record?.appearance||base.appearance||null,level:record?.level??base.level??70,class:record?.className||base.class||'',armoryUpdatedAt:record?.updatedAt||new Date().toISOString(),__wrInspectSnapshot:true};}
  function publish(scope,p){if(!scope?.isConnected||scope!==activeScope||!p||key(p.name)!==key(activeName))return;window.WarRoomInspectLive={scope,character:p,name:p.name,gear:gearRows(p),publishedAt:Date.now(),snapshotKey:key(p.name)};window.dispatchEvent(new CustomEvent('warroom:inspect-character-ready',{detail:{scope,name:p.name,character:p}}));}
  function setMetric(scope,label,value){const lab=findExact(scope,label),box=lab?.parentElement;if(!box||!scope.contains(box))return;const target=[...box.children].find(x=>x!==lab&&/^(DIV|SPAN|B|STRONG|P)$/i.test(x.tagName));if(target&&target.textContent!==String(value))target.textContent=value}
  function render(scope,p){if(!scope?.isConnected||scope!==activeScope||!p||key(p.name)!==key(activeName))return false;const spec=p.spec||p.talentSpec||p.activeSpec||'Unknown';setMetric(scope,'ROLE',spec);setMetric(scope,'ROLE / SPEC',spec);setMetric(scope,'AVG ILVL',p.ilvl??'—');const tl=findExact(scope,'TALENT BUILD');if(tl?.parentElement){const v=[...tl.parentElement.children].find(x=>x!==tl);if(v)v.textContent=talentText(p)}const eq=findExact(scope,'EQUIPMENT');if(eq?.parentElement){const v=[...eq.parentElement.children].find(x=>x!==eq),g=gearRows(p);if(v)v.textContent=g.length?`${g.length} equipped items loaded from armory.`:'No verified equipment returned.'}publish(scope,p);return true}
  async function fetchSnapshot(scope,name,token){const k=key(name);let job=inflight.get(k);if(!job){job=(async()=>{const api=window.WarRoomArmoryRefresh;if(!api?.refreshNames)throw new Error('Armory refresh service unavailable');const result=await api.refreshNames([name],{concurrency:1,deadlineMs:10000});const record=(result?.records||[]).find(r=>key(r?.name)===k);if(!record)throw new Error('No matching armory record returned for '+name);const snap=snapshotFromRecord(record,name);snapshots.set(k,snap);return snap})().finally(()=>inflight.delete(k));inflight.set(k,job)}try{const snap=await job;if(token!==requestToken||scope!==activeScope||key(name)!==key(activeName))return;if(key(snap.name)!==key(name))throw new Error('Inspect snapshot identity mismatch');render(scope,snap)}catch(e){if(token!==requestToken||scope!==activeScope)return;setMetric(scope,'ROLE / SPEC','Armory unavailable');setMetric(scope,'AVG ILVL','—');console.warn('War Room Inspect snapshot:',e)}}
